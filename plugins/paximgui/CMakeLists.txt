#### Dear ImGui (vendored) ####

find_package(SDL2 REQUIRED)
find_package(OpenGL REQUIRED)
find_package(GLEW REQUIRED)
find_package(antlr4-runtime REQUIRED)

add_library(imgui STATIC
        imgui/imgui.cpp
        imgui/imgui_demo.cpp
        imgui/imgui_draw.cpp
        imgui/imgui_tables.cpp
        imgui/imgui_widgets.cpp
        imgui/imgui_impl_opengl3.cpp
        imgui/imgui_impl_sdl.cpp)
target_include_directories(imgui PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/imgui>)
target_link_libraries(imgui PUBLIC SDL2::SDL2 OpenGL::GL GLEW::GLEW)

#### Parser for the shader annotations (ANTLR sources are generated by genImguiPPAntlrFiles.sh and checked in) ####

add_library(imguipp STATIC
        imguipp_antlr/ImguiPPLexer.cpp
        imguipp_antlr/ImguiPPParser.cpp
        imguipp_antlr/ImguiPPParserBaseListener.cpp
        imguipp_antlr/ImguiPPParserBaseVisitor.cpp
        imguipp_antlr/ImguiPPParserListener.cpp
        imguipp_antlr/ImguiPPParserVisitor.cpp)
# The ANTLR runtime package only sets the variable ANTLR4_INCLUDE_DIR; its target has no include directories.
# The installed PaxEngine3 package adds them for consumers (see PaxEngine3Config.cmake.in).
target_include_directories(imguipp PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/imguipp_antlr>
        $<BUILD_INTERFACE:${ANTLR4_INCLUDE_DIR}>)
target_link_libraries(imguipp PUBLIC antlr4_static)

#### Plugin ####

set(PAX_IMGUI_SOURCE_FILES
        src/ImguiPlugin.cpp
        src/ImguiRenderer.cpp
        src/Variable.cpp
        src/ImguiUniformDetector.cpp
        src/ImguiUniformDetectorVisitor.cpp)

add_library(paximgui ${PAX_IMGUI_SOURCE_FILES})
add_library(PaxEngine3::imgui ALIAS paximgui)
target_include_directories(paximgui PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(paximgui PUBLIC paxcore paxsdl paxopengl imgui imguipp)

# Their export names must not clash with the plugin PaxEngine3::imgui.
set_target_properties(imgui PROPERTIES EXPORT_NAME imgui_lib)
set_target_properties(imguipp PROPERTIES EXPORT_NAME imguipp)

# imgui and imguipp are linked statically into paximgui, so they have to be exported as well.
set(PAX_PLUGIN_EXTRA_TARGETS imgui imguipp PARENT_SCOPE)
