# Demo games. Each one is only built when the plugins it needs are part of the build.
set(PAX_DEMO_REQUIREMENTS_PlatformerDemo sdl opengl physics tiles)
set(PAX_DEMO_REQUIREMENTS_TileDemo sdl opengl physics tiles)
set(PAX_DEMO_REQUIREMENTS_PlaygroundDemo sdl opengl assimp)
set(PAX_DEMO_REQUIREMENTS_RayMarcher sdl opengl)

foreach (demo PlatformerDemo TileDemo PlaygroundDemo RayMarcher)
    set(demo_buildable TRUE)
    foreach (plugin IN LISTS PAX_DEMO_REQUIREMENTS_${demo})
        if (NOT plugin IN_LIST PAX_PLUGINS)
            set(demo_buildable FALSE)
        endif()
    endforeach()
    if (demo_buildable)
        add_subdirectory(${demo})
    else()
        message(STATUS "PaxEngine3: skipping demo ${demo} because a required plugin is missing (${PAX_DEMO_REQUIREMENTS_${demo}})")
    endif()
endforeach()
